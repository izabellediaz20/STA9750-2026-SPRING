state_to_state_migration_2024 <- function() {

  file_path <- "data/mp03/State_to_State_Migration_Table_2024_T13.xlsx"

  if (!file.exists(file_path)) {
    download.file(
      "https://www2.census.gov/programs-surveys/demo/tables/geographic-mobility/2024/state-to-state-migration/State_to_State_Migration_Table_2024_T13.xlsx",
      file_path,
      mode = "wb"
    )
  }

  state_lookup <- c(
    setNames(state.abb, state.name),
    "District of Columbia" = "DC"
  )

  movers_2024 <- read_excel(
    file_path,
    sheet = "Table",
    skip = 8,
    col_names = c("state_current", "state_1y", "population", "moe")
  ) %>%
    mutate(
      state_current = str_trim(as.character(state_current)),
      state_1y = str_trim(as.character(state_1y)),
      population = str_trim(as.character(population)),
      population = replace_na(population, "0"),
      population = replace(population, population == "N", "0")
    ) %>%
    filter(
      !is.na(state_current),
      !is.na(state_1y),
      population != "X",
      state_current != "District of Columbia",
      state_1y != "District of Columbia"
    ) %>%
    mutate(
      population = suppressWarnings(as.numeric(population)),
      year_current = 2024,
      year_1y = 2023,
      state_current = unname(state_lookup[state_current]),
      state_1y = unname(state_lookup[state_1y])
    ) %>%
    filter(
      !is.na(population),
      !is.na(state_current),
      !is.na(state_1y)
    ) %>%
    select(state_current, state_1y, population, year_current, year_1y)

  stationary_2024 <- read_excel(
    file_path,
    sheet = "Supplemental - Current Res",
    skip = 10,
    col_names = FALSE
  ) %>%
    select(
      state_current = 1,
      same_residence = 4,
      same_state = 6
    ) %>%
    filter(!is.na(state_current), state_current != "District of Columbia") %>%
    mutate(
      state_current = str_trim(as.character(state_current)),
      state_1y = state_current,
      same_residence = suppressWarnings(as.numeric(same_residence)),
      same_state = suppressWarnings(as.numeric(same_state)),
      population = replace_na(same_residence, 0) + replace_na(same_state, 0),
      year_current = 2024,
      year_1y = 2023,
      state_current = unname(state_lookup[state_current]),
      state_1y = unname(state_lookup[state_1y])
    ) %>%
    filter(
      !is.na(state_current),
      !is.na(state_1y)
    ) %>%
    select(state_current, state_1y, population, year_current, year_1y)

  bind_rows(movers_2024, stationary_2024)
}

state_to_state_migration_2023 <- function() {

  file_path <- "data/mp03/State_to_State_Migration_Table_2023_T13.xlsx"

  if (!file.exists(file_path)) {
    download.file(
      "https://www2.census.gov/programs-surveys/demo/tables/geographic-mobility/2023/state-to-state-migration/State_to_State_Migration_Table_2023_T13.xlsx",
      file_path,
      mode = "wb"
    )
  }

  states_with_dc <- c(
    state.name[1:8],
    "District of Columbia",
    state.name[9:50]
  )

  state_lookup <- c(
    setNames(state.abb, state.name),
    "District of Columbia" = "DC"
  )

  stayed_data <- read_excel(
    file_path,
    range = readxl::cell_cols("A:F"),
    col_names = FALSE
  ) %>%
    slice(-(1:8)) %>%
    select(
      state_to = 1,
      same_house = 4,
      same_state = 6
    ) %>%
    filter(!is.na(state_to)) %>%
    mutate(
      state_to = str_trim(as.character(state_to)),
      same_house = suppressWarnings(as.numeric(same_house)),
      same_state = suppressWarnings(as.numeric(same_state))
    ) %>%
    filter(state_to %in% state.name) %>%
    mutate(
      state_from = state_to,
      population = replace_na(same_house, 0) + replace_na(same_state, 0),
      year_current = 2023,
      year_1y = 2022,
      state_to = unname(state_lookup[state_to]),
      state_from = unname(state_lookup[state_from])
    ) %>%
    filter(state_to != "DC", state_from != "DC") %>%
    select(state_to, state_from, population, year_current, year_1y)

  flow_data <- read_excel(
    file_path,
    range = readxl::cell_cols("J:DG"),
    col_names = FALSE
  ) %>%
    slice(1:51) %>%
    select(1:51) %>%
    setNames(states_with_dc) %>%
    mutate(state_to = states_with_dc) %>%
    pivot_longer(
      cols = -state_to,
      names_to = "state_from",
      values_to = "population"
    ) %>%
    mutate(
      state_to = str_trim(as.character(state_to)),
      state_from = str_trim(as.character(state_from)),
      population = str_trim(as.character(population))
    ) %>%
    filter(
      !is.na(population),
      population != "N",
      population != "X",
      state_to != "District of Columbia",
      state_from != "District of Columbia",
      state_to != state_from
    ) %>%
    mutate(
      population = suppressWarnings(as.numeric(population)),
      year_current = 2023,
      year_1y = 2022,
      state_to = unname(state_lookup[state_to]),
      state_from = unname(state_lookup[state_from])
    ) %>%
    filter(
      !is.na(population),
      !is.na(state_to),
      !is.na(state_from)
    ) %>%
    select(state_to, state_from, population, year_current, year_1y)

  bind_rows(stayed_data, flow_data) %>%
    rename(
      state_current = state_to,
      state_1y = state_from
    )
}

To construct the 2023 state-to-state migration dataset, I imported the within-state counts and the interstate flow section separately. I then cleaned the data, converted it into a long format, removed District of Columbia observations, and combined the pieces into one tidy table.

metro_to_metro_migration <- function() {
  library(tidyverse)
  library(jsonlite)
  
  dir.create("data/mp03", recursive = TRUE, showWarnings = FALSE)
  
  file_path <- "data/mp03/metro_to_metro_migration_2023.json"
  
  api_url <- paste0(
    "https://api.census.gov/data/2020/acs/flows?",
    "get=MOVEDIN,MOVEDOUT,FULL1_NAME,FULL2_NAME,GEOID1,GEOID2&",
    "for=metropolitan%20statistical%20area/micropolitan%20statistical%20area:*"
  )
  
  if (!file.exists(file_path)) {
    download.file(
      url = api_url,
      destfile = file_path,
      mode = "wb"
    )
  }
  
  move_row_to_colnames <- function(X, row = 1) {
    X <- as.data.frame(X, stringsAsFactors = FALSE)
    Xrow <- X[row, , drop = FALSE]
    X <- X[-row, , drop = FALSE]
    colnames(X) <- unlist(Xrow[1, ])
    tibble::as_tibble(X) |>
      readr::type_convert()
  }
  
  pull_state_from_metro <- function(metro_name) {
    stringr::str_extract(metro_name, "(?<=, )[A-Z]{2}(?=-[A-Z]{2}|- Metro Area)")
  }
  
  metro_data <- read_json(file_path, simplifyVector = TRUE) |>
    move_row_to_colnames() |>
    transmute(
      metro_current = FULL1_NAME,
      metro_1y = FULL2_NAME,
      moved_in = MOVEDIN,
      moved_out = MOVEDOUT,
      state_current = pull_state_from_metro(metro_current),
      state_1y = pull_state_from_metro(metro_1y)
    )
  
  metro_data
}

To import metro-to-metro migration flows, I created a function called metro_to_metro_migration() that retrieves data from the Census API and saves it locally as a JSON file. This ensures the data is only downloaded once and can be reused efficiently.

Because the raw API output does not include proper column names, I used a helper function (move_row_to_colnames) to convert the first row of the dataset into column names and then cleaned the data structure. I also extracted state abbreviations from metro area names using a regular expression.

The final dataset includes metro area names, in-migration counts, out-migration counts, and associated state abbreviations, making it suitable for further analysis.

Task 8: Reapportionment

To estimate congressional districts from the projected 2030 state populations, I applied the Huntington-Hill method. This method allocates seats by calculating priority values for each state based on population and then assigning seats until all 435 congressional districts are distributed.

NoteShow Code 
library(tidyverse)
library(gt)

MAX_DISTRICTS <- 100
N_DISTRICTS <- 435

apportionment_2030 <- pop_projection_2030 |>
  transmute(
    state,
    population = as.integer(round(population))
  ) |>
  tidyr::expand_grid(cd = seq(0, MAX_DISTRICTS)) |>
  mutate(
    hh_den = sqrt(cd * (cd + 1)),
    hh_weight = population / hh_den
  ) |>
  slice_max(hh_weight, n = N_DISTRICTS) |>
  group_by(state) |>
  summarise(
    population = first(population),
    n_districts = n(),
    .groups = "drop"
  ) |>
  mutate(
    pop_per_district = population / n_districts
  ) |>
  arrange(desc(n_districts), desc(population), state)
